## 一、项目概述

public Wallet createDreamWallet(Long userId, Integer dreamType, String walletName,
                                BigDecimal targetAmount, Long itemId, Integer days) {
    // 1. 获取商品信息
    DreamItem dreamItem = dreamItemMapper.selectById(itemId);

    // 2. 创建钱包
    Wallet wallet = new Wallet();
    wallet.setUserId(userId);
    wallet.setName(walletName);
    wallet.setType(4); // 4-梦想攒钱(购物)
    wallet.setDreamType(1);
    wallet.setDreamTargetAmount(targetAmount);
    wallet.setDreamItemName(dreamItem.getName());

    // 3. 从配置获取初始进度图片（第1张，0-20%）
    List<String> progressImages = dreamImageConfigService.getProgressImages(1, dreamItem.getName());
    wallet.setDreamItemImage(progressImages.get(0));

    // 4. 创建默认奖励（5个里程碑）
    dreamRewardMapper.createDefaultRewards(wallet.getId());

    return wallet;

public boolean updateDreamProgress(Long walletId, BigDecimal newBalance) {
    Wallet wallet = walletMapper.selectById(walletId);

    // 计算新进度
    Integer newProgress = (newBalance / wallet.getDreamTargetAmount() * 100).intValue();
    Integer oldProgress = wallet.getDreamProgress();

    wallet.setDreamProgress(newProgress);

    // 检查是否达到新里程碑（20%, 40%, 60%, 80%, 100%）
    for (int milestone : Arrays.asList(20, 40, 60, 80, 100)) {
        if (oldProgress < milestone && newProgress >= milestone) {
            // 解锁对应的奖励
            dreamRewardMapper.unlockReward(walletId, milestone);

            // 记录进度历史
            DreamProgressHistory history = new DreamProgressHistory();
            history.setWalletId(walletId);
            history.setProgress(milestone);
            history.setTriggerType(1); // 1-存入
            dreamProgressHistoryMapper.insert(history);
        }
    }

    walletMapper.updateById(wallet);
    return true;
}
```

    DreamItem dreamItem = dreamItemMapper.selectById(itemId);

    Wallet wallet = new Wallet();

                                     days,

    wallet.setDreamItinerary(itinerary);
    walletMapper.updateById(wallet);

@EventListener
public void onTransactionCreated(TransactionCreatedEvent event) {
    Transaction transaction = event.getTransaction();

    // 获取钱包的搭子列表
    List<BuddyCharacter> buddies = buddyService.getBuddyCharactersByWalletId(
        transaction.getWalletId()
    );

    // 随机选择1-2个搭子进行评论
    int commentCount = RandomUtils.nextInt(1, 3);
    Collections.shuffle(buddies);

    for (int i = 0; i < commentCount && i < buddies.size(); i++) {
        BuddyCharacter buddy = buddies.get(i);

        // 生成符合角色性格的评论
        String content = generateBuddyComment(buddy, transaction);

        // 生成语音（TTS）
        String voiceUrl = ttsService.generateVoice(content, buddy.getVoiceType());

        // 保存评论
        PostComment comment = new PostComment();
        comment.setPostId(transaction.getId());
        comment.setUserId(buddy.getId());
        comment.setContent(content);
        comment.setVoiceUrl(voiceUrl);
        comment.setBuddyCharacterId(buddy.getId());
        postCommentMapper.insert(comment);
    }
}

// ScriptService.java
public void makeChoice(Long userId, Long walletId, Long scriptId,
                      Integer currentChapter, String selectedChoice, BigDecimal amount) {
    // 1. 扣除钱包金额
    walletService.scriptSaving(walletId, amount,
        String.format("剧本《%s》第%d章选择", scriptId, currentChapter)
    );

    // 2. 更新用户剧本进度
    UserScriptProgress progress = userScriptProgressMapper.selectByUserAndScript(userId, scriptId);
    progress.setCurrentChapter(currentChapter + 1);
    progress.setSelectedChoice(selectedChoice);
    userScriptProgressMapper.updateById(progress);

    // 3. 检查是否完成剧本
    Script script = scriptMapper.selectById(scriptId);
    if (progress.getCurrentChapter() >= script.getTotalChapters()) {
        progress.setCompleted(true);
        progress.setCompletedTime(LocalDateTime.now());
    }

    userScriptProgressMapper.updateById(progress);
}
```

    Map<String, Object> requestBody = new HashMap<>();
    requestBody.put("text", text);
    requestBody.put("voice_type", voiceType); // BV001_streaming等

    requestBody.put("volume", 1.0);

public BigDecimal extractAmountFromImage(String imageUrl) {
    // 1. 下载图片
    byte[] imageData = httpUtil.getBytes(imageUrl);

    // 2. 调用字节OCR API
    String ocrResult = ocrService.recognizeText(imageData);

    // 3. 正则提取金额
    Pattern pattern = Pattern.compile("\\d+(\\.\\d{1,2})?");
    Matcher matcher = pattern.matcher(ocrResult);

    if (matcher.find()) {
        return new BigDecimal(matcher.group());
    }

    return BigDecimal.ZERO;
}

// const BASE_URL = 'https://heartllo.cn/api'  // 生产环境

// 通用请求函数
function request(options) {
  return new Promise((resolve, reject) => {
    const token = wx.getStorageSync('token')

    const header = {
      'Content-Type': 'application/json; charset=utf-8',
      ...options.header
    }

    if (token) {
      header['Authorization'] = `Bearer ${token}`
    }

    wx.request({
      url: `${BASE_URL}${options.url}`,
      method: options.method || 'GET',
      data: options.data,
      header: header,
      success: (res) => {
        if (res.statusCode === 200 && res.data.code === 200) {
          resolve(res.data)
        } else if (res.statusCode === 401) {
          // token过期，跳转登录
          wx.redirectTo({ url: '/pages/login/login' })
          reject(new Error('登录已过期'))
        } else {
          reject(new Error(res.data.message || '请求失败'))
        }
      },
      fail: (error) => {
        reject(new Error('网络连接失败'))
      }
    })
  })
}

public class Result<T> {
    private Integer code;      // 200-成功，其他-失败
    private String message;    // 返回消息
    private T data;           // 返回数据
    private String timestamp; // 时间戳

        Result<T> result = new Result<>();
        result.setCode(200);
        result.setMessage(message);
        result.setData(data);
        result.setSuccess(true);
        result.setTimestamp(String.valueOf(System.currentTimeMillis()));

        Result<T> result = new Result<>();
        result.setCode(500);
        result.setMessage(message);
        result.setSuccess(false);
        result.setTimestamp(String.valueOf(System.currentTimeMillis()));
        return result;
    }
}

public class JwtAuthenticationFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                   HttpServletResponse response,
                                   FilterChain filterChain) {
        String token = request.getHeader("Authorization");

        if (token != null && token.startsWith("Bearer ")) {
            token = token.substring(7);

            try {
                Long userId = JwtUtil.parseToken(token);
                // 将userId设置到SecurityContext
                SecurityContextHolder.getContext().setAuthentication(
                    new UsernamePasswordAuthenticationToken(userId, null, null)
                );
            } catch (Exception e) {
                response.setStatus(HttpStatus.UNAUTHORIZED.value());
                return;
            }
        }

        filterChain.doFilter(request, response);
    }
}

## 七、业务流程

### 7.1 用户注册登录流程

```
1. 用户打开小程序
   ↓
2. 调用 wx.login() 获取微信code
   ↓
3. 前端调用 /api/auth/wechat-login
   ↓
4. 后端用code换取openid
   ↓
5. 检查openid是否已注册
   ├─ 已注册：返回token
   └─ 未注册：创建新用户 → 返回token
   ↓
6. 前端保存token到storage
   ↓
7. 跳转到主页
```

### 7.2 创建梦想钱包流程（旅行）

```
1. 用户点击"新建钱兜兜"
   ↓
2. 选择"梦想攒"
   ↓
3. 选择"旅行"类型
   ↓
4. 浏览旅行目的地列表
   ├─ GET /api/dream-wallet/items?category=2
   └─ 返回：北京、西安、拉萨、悉尼等
   ↓
5. 选择目的地（如：阿格拉）
   ↓
6. 设置旅行天数和预算
   ↓
7. 提交创建
   ├─ POST /api/dream-wallet/create
   └─ 参数：{userId, dreamType: 2, walletName, targetAmount, itemId, days}
   ↓
8. 后端创建钱包
   ├─ 设置 type = 5
   ├─ 设置 dreamType = 2
   ├─ 设置 dreamDestination = "agra"
   ├─ 从 dream_items 获取 image_url
   ├─ 设置 dreamItemImage = "https://...阿格拉_back.jpg"
   └─ 异步生成AI旅行行程
   ↓
9. 创建默认奖励（5个里程碑）
   ↓
10. 返回钱包信息
    ↓
11. 前端跳转到梦想详情页
```

### 7.3 转入资金并触发AI评论流程

```
1. 用户在钱包详情页点击"转入"
   ↓
2. 输入金额、描述、上传图片
   ↓
3. 提交转入
   ├─ POST /api/wallet/transfer-in
   └─ 参数：{walletId, amount, description, imageUrl, note}
   ↓
4. 后端处理转入
   ├─ 更新钱包余额
   ├─ 创建交易记录
   └─ 发布 TransactionCreatedEvent 事件
   ↓
5. 事件监听器异步处理
   ├─ AiLoverInteractionEventListener（情侣钱包）
   │   ├─ 获取AI伴侣信息
   │   ├─ 生成AI评论内容
   │   ├─ 调用TTS生成语音
   │   └─ 保存评论
   │
   └─ BuddyAutoCommentService（搭子钱包）
       ├─ 获取钱包搭子列表
       ├─ 随机选择1-2个搭子
       ├─ 生成符合角色性格的评论
       ├─ 调用TTS生成语音
       └─ 保存评论
   ↓
6. 梦想钱包检查进度
   ├─ 计算新进度：newProgress = (balance / targetAmount) * 100
   ├─ 检查是否达到里程碑（20%, 40%, 60%, 80%, 100%）
   └─ 解锁对应奖励
   ↓
7. 返回交易成功
   ↓
8. 前端刷新钱包详情
```

### 7.4 梦想钱包进度更新流程

```
购物钱包（type=4, dreamType=1）：
1. 用户存入资金
   ↓
2. 计算进度：0% → 25%
   ↓
3. 后端 getCurrentStageImage()
   ├─ 调用 dreamImageConfigService.getProgressImages(1, "car")
   ├─ 返回5张进度图片
   └─ 根据进度选择：part2.png（20-40%）
   ↓
4. 更新 wallet.dreamItemImage = part2.png
   ↓
5. 前端显示新的进度图片（汽车从轮廓到完整）

---

旅行钱包（type=5, dreamType=2）：
1. 用户存入资金
   ↓
2. 计算进度：0% → 25%
   ↓
3. 后端 getCurrentStageImage()
   └─ 直接返回 wallet.dreamItemImage（固定的目的地背景图）
   ↓
4. 不更新图片
   ↓
5. 前端始终显示相同的目的地背景图
```

### 8.6 JWT认证机制

**Token生成**：
```java
public class JwtUtil {
    private static final String SECRET = "your-secret-key";
    private static final long EXPIRE_TIME = 7 * 24 * 60 * 60 * 1000; // 7天

    public static String generateToken(Long userId) {
        return JWT.create()
            .withClaim("userId", userId)
            .withExpiresAt(new Date(System.currentTimeMillis() + EXPIRE_TIME))
            .sign(Algorithm.HMAC256(SECRET));
    }

    public static Long parseToken(String token) {
        DecodedJWT jwt = JWT.require(Algorithm.HMAC256(SECRET))
            .build()
            .verify(token);
        return jwt.getClaim("userId").asLong();
    }
}
```
